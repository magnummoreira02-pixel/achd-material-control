import React, { useState, useMemo, useRef, useCallback } from "react";
import * as XLSX from "xlsx";
import {
  UploadCloud,
  FileSpreadsheet,
  Settings2,
  Search,
  X,
  Check,
  RotateCcw,
  AlertTriangle,
} from "lucide-react";

const FONT_IMPORT =
  "@import url('https://fonts.googleapis.com/css2?family=Space+Grotesk:wght@500;600;700&family=IBM+Plex+Sans:wght@400;500;600&family=IBM+Plex+Mono:wght@400;500;600&display=swap');";

const PAPER = "#F1F1EC";
const PANEL = "#FFFFFF";
const INK = "#1E2A38";
const INK_SOFT = "#5B6B7C";
const LINE = "#D8D9D0";
const LINE_STRONG = "#B7BAAF";
const AMBER = "#C97A1C";
const AMBER_BG = "#FBEEDD";
const GREEN = "#2E7D5B";
const GREEN_BG = "#E4F2EA";
const RED = "#B3402E";
const RED_BG = "#FBEAE6";

function guessIdColumn(headers) {
  const patterns = ["qr", "codigo", "código", "cod", "id", "sku", "material", "tag"];
  for (const p of patterns) {
    const found = headers.find((h) => h.toLowerCase().includes(p));
    if (found) return found;
  }
  return headers[0] || "";
}

function StepBadge({ n, active }) {
  return (
    <div
      style={{
        width: 30,
        height: 30,
        borderRadius: 4,
        border: `1.5px solid ${active ? INK : LINE_STRONG}`,
        color: active ? PAPER : INK_SOFT,
        background: active ? INK : "transparent",
        display: "flex",
        alignItems: "center",
        justifyContent: "center",
        fontFamily: "'IBM Plex Mono', monospace",
        fontWeight: 600,
        fontSize: 13,
        flexShrink: 0,
      }}
    >
      {n}
    </div>
  );
}

function Panel({ step, title, active, children, trailing }) {
  return (
    <section
      style={{
        background: PANEL,
        border: `1px solid ${LINE}`,
        borderRadius: 6,
        marginBottom: 20,
        opacity: active ? 1 : 0.55,
        transition: "opacity 0.2s ease",
      }}
    >
      <div
        style={{
          display: "flex",
          alignItems: "center",
          gap: 12,
          padding: "14px 18px",
          borderBottom: `1px solid ${LINE}`,
        }}
      >
        <StepBadge n={step} active={active} />
        <span
          style={{
            fontFamily: "'Space Grotesk', sans-serif",
            fontWeight: 600,
            fontSize: 14,
            letterSpacing: "0.03em",
            textTransform: "uppercase",
            color: INK,
            flex: 1,
          }}
        >
          {title}
        </span>
        {trailing}
      </div>
      <div style={{ padding: 18 }}>{children}</div>
    </section>
  );
}

export default function App() {
  const [fileName, setFileName] = useState("");
  const [headers, setHeaders] = useState([]);
  const [rows, setRows] = useState([]);
  const [idColumn, setIdColumn] = useState("");
  const [displayColumns, setDisplayColumns] = useState([]);
  const [query, setQuery] = useState("");
  const [matched, setMatched] = useState(null);
  const [searchState, setSearchState] = useState("idle"); // idle | found | notfound
  const [dragOver, setDragOver] = useState(false);
  const [parseError, setParseError] = useState("");
  const inputRef = useRef(null);

  const hasData = rows.length > 0 && headers.length > 0;
  const readyToSearch = hasData && idColumn;

  const processFile = useCallback((file) => {
    setParseError("");
    const reader = new FileReader();
    reader.onload = (e) => {
      try {
        const data = new Uint8Array(e.target.result);
        const workbook = XLSX.read(data, { type: "array" });
        const sheetName = workbook.SheetNames[0];
        const sheet = workbook.Sheets[sheetName];
        const json = XLSX.utils.sheet_to_json(sheet, { defval: "" });
        if (!json.length) {
          setParseError("A planilha está vazia ou não foi possível ler as colunas.");
          return;
        }
        const hdrs = Object.keys(json[0]);
        setHeaders(hdrs);
        setRows(json);
        setFileName(file.name);
        const guessed = guessIdColumn(hdrs);
        setIdColumn(guessed);
        setDisplayColumns(hdrs.filter((h) => h !== guessed));
        setQuery("");
        setMatched(null);
        setSearchState("idle");
      } catch (err) {
        setParseError("Não foi possível ler este arquivo. Confirme se é um .xlsx ou .csv válido.");
      }
    };
    reader.readAsArrayBuffer(file);
  }, []);

  const handleFileInput = (e) => {
    const file = e.target.files && e.target.files[0];
    if (file) processFile(file);
  };

  const handleDrop = (e) => {
    e.preventDefault();
    setDragOver(false);
    const file = e.dataTransfer.files && e.dataTransfer.files[0];
    if (file) processFile(file);
  };

  const resetAll = () => {
    setFileName("");
    setHeaders([]);
    setRows([]);
    setIdColumn("");
    setDisplayColumns([]);
    setQuery("");
    setMatched(null);
    setSearchState("idle");
    setParseError("");
    if (inputRef.current) inputRef.current.value = "";
  };

  const suggestions = useMemo(() => {
    if (!readyToSearch || !query.trim()) return [];
    const q = query.trim().toLowerCase();
    return rows
      .filter((r) => String(r[idColumn] ?? "").toLowerCase().includes(q))
      .slice(0, 8);
  }, [rows, idColumn, query, readyToSearch]);

  const runSearch = (value) => {
    const q = String(value ?? query).trim().toLowerCase();
    if (!q) {
      setMatched(null);
      setSearchState("idle");
      return;
    }
    const exact = rows.find((r) => String(r[idColumn] ?? "").trim().toLowerCase() === q);
    if (exact) {
      setMatched(exact);
      setSearchState("found");
    } else {
      setMatched(null);
      setSearchState("notfound");
    }
  };

  const toggleColumn = (h) => {
    setDisplayColumns((prev) =>
      prev.includes(h) ? prev.filter((c) => c !== h) : [...prev, h]
    );
  };

  return (
    <div
      style={{
        fontFamily: "'IBM Plex Sans', sans-serif",
        background: PAPER,
        minHeight: "100%",
        padding: "32px 20px",
        color: INK,
      }}
    >
      <style>{FONT_IMPORT}</style>
      <div style={{ maxWidth: 720, margin: "0 auto" }}>
        <div style={{ marginBottom: 28 }}>
          <div
            style={{
              fontFamily: "'IBM Plex Mono', monospace",
              fontSize: 12,
              letterSpacing: "0.12em",
              color: AMBER,
              fontWeight: 600,
              marginBottom: 6,
            }}
          >
            CONSOLE DE BUSCA
          </div>
          <h1
            style={{
              fontFamily: "'Space Grotesk', sans-serif",
              fontSize: 32,
              fontWeight: 700,
              margin: 0,
              letterSpacing: "-0.01em",
            }}
          >
            Seleção de material
          </h1>
          <p style={{ color: INK_SOFT, fontSize: 14, marginTop: 8, lineHeight: 1.6 }}>
            Importe uma planilha, defina qual coluna guarda o número ou código
            QR, e busque um item digitando ou lendo o código.
          </p>
        </div>

        {/* STEP 1 — IMPORT */}
        <Panel
          step={1}
          title="Importar planilha"
          active={true}
          trailing={
            hasData && (
              <button
                onClick={resetAll}
                style={{
                  display: "flex",
                  alignItems: "center",
                  gap: 6,
                  background: "transparent",
                  border: `1px solid ${LINE_STRONG}`,
                  borderRadius: 4,
                  padding: "6px 10px",
                  fontSize: 12,
                  color: INK_SOFT,
                  cursor: "pointer",
                  fontFamily: "'IBM Plex Sans', sans-serif",
                }}
              >
                <RotateCcw size={13} /> Trocar arquivo
              </button>
            )
          }
        >
          {!hasData ? (
            <div
              onDragOver={(e) => {
                e.preventDefault();
                setDragOver(true);
              }}
              onDragLeave={() => setDragOver(false)}
              onDrop={handleDrop}
              onClick={() => inputRef.current && inputRef.current.click()}
              style={{
                border: `2px dashed ${dragOver ? AMBER : LINE_STRONG}`,
                borderRadius: 6,
                padding: "36px 20px",
                textAlign: "center",
                cursor: "pointer",
                background: dragOver ? AMBER_BG : PAPER,
                transition: "all 0.15s ease",
              }}
            >
              <UploadCloud
                size={28}
                color={dragOver ? AMBER : INK_SOFT}
                style={{ marginBottom: 10 }}
              />
              <div style={{ fontWeight: 600, fontSize: 14, marginBottom: 4 }}>
                Arraste a planilha aqui ou clique para escolher
              </div>
              <div style={{ fontSize: 12.5, color: INK_SOFT }}>
                Formatos aceitos: .xlsx, .xls, .csv
              </div>
              <input
                ref={inputRef}
                type="file"
                accept=".xlsx,.xls,.csv"
                onChange={handleFileInput}
                style={{ display: "none" }}
              />
            </div>
          ) : (
            <div style={{ display: "flex", alignItems: "center", gap: 10 }}>
              <FileSpreadsheet size={20} color={GREEN} />
              <div style={{ flex: 1 }}>
                <div style={{ fontWeight: 600, fontSize: 13.5 }}>{fileName}</div>
                <div style={{ fontSize: 12, color: INK_SOFT }}>
                  {rows.length} {rows.length === 1 ? "linha" : "linhas"} · {headers.length}{" "}
                  {headers.length === 1 ? "coluna" : "colunas"}
                </div>
              </div>
            </div>
          )}
          {parseError && (
            <div
              style={{
                marginTop: 12,
                display: "flex",
                gap: 8,
                alignItems: "flex-start",
                background: RED_BG,
                color: RED,
                border: `1px solid ${RED}`,
                borderRadius: 4,
                padding: "10px 12px",
                fontSize: 12.5,
              }}
            >
              <AlertTriangle size={15} style={{ flexShrink: 0, marginTop: 1 }} />
              {parseError}
            </div>
          )}
        </Panel>

        {/* STEP 2 — CONFIG */}
        <Panel step={2} title="Configurar colunas" active={hasData}>
          {!hasData ? (
            <div style={{ fontSize: 13, color: INK_SOFT }}>
              Importe uma planilha para configurar as colunas.
            </div>
          ) : (
            <div style={{ display: "flex", flexDirection: "column", gap: 18 }}>
              <div>
                <label
                  style={{
                    display: "flex",
                    alignItems: "center",
                    gap: 6,
                    fontSize: 12.5,
                    fontWeight: 600,
                    color: INK_SOFT,
                    textTransform: "uppercase",
                    letterSpacing: "0.04em",
                    marginBottom: 8,
                  }}
                >
                  <Settings2 size={13} /> Coluna com o número / código QR
                </label>
                <select
                  value={idColumn}
                  onChange={(e) => {
                    setIdColumn(e.target.value);
                    setDisplayColumns(headers.filter((h) => h !== e.target.value));
                    setQuery("");
                    setMatched(null);
                    setSearchState("idle");
                  }}
                  style={{
                    width: "100%",
                    padding: "9px 10px",
                    borderRadius: 4,
                    border: `1px solid ${LINE_STRONG}`,
                    background: PAPER,
                    fontFamily: "'IBM Plex Mono', monospace",
                    fontSize: 13,
                    color: INK,
                  }}
                >
                  {headers.map((h) => (
                    <option key={h} value={h}>
                      {h}
                    </option>
                  ))}
                </select>
              </div>

              <div>
                <div
                  style={{
                    fontSize: 12.5,
                    fontWeight: 600,
                    color: INK_SOFT,
                    textTransform: "uppercase",
                    letterSpacing: "0.04em",
                    marginBottom: 8,
                  }}
                >
                  Colunas a exibir no resultado
                </div>
                <div style={{ display: "flex", flexWrap: "wrap", gap: 8 }}>
                  {headers
                    .filter((h) => h !== idColumn)
                    .map((h) => {
                      const checked = displayColumns.includes(h);
                      return (
                        <button
                          key={h}
                          onClick={() => toggleColumn(h)}
                          style={{
                            display: "flex",
                            alignItems: "center",
                            gap: 6,
                            padding: "6px 10px",
                            borderRadius: 4,
                            border: `1px solid ${checked ? INK : LINE_STRONG}`,
                            background: checked ? INK : "transparent",
                            color: checked ? PAPER : INK_SOFT,
                            fontSize: 12.5,
                            cursor: "pointer",
                            fontFamily: "'IBM Plex Sans', sans-serif",
                          }}
                        >
                          {checked && <Check size={12} />}
                          {h}
                        </button>
                      );
                    })}
                </div>
              </div>
            </div>
          )}
        </Panel>

        {/* STEP 3 — SEARCH */}
        <Panel step={3} title="Buscar material" active={readyToSearch}>
          {!readyToSearch ? (
            <div style={{ fontSize: 13, color: INK_SOFT }}>
              Configure a coluna de busca para habilitar a leitura.
            </div>
          ) : (
            <div>
              <div style={{ position: "relative", padding: "14px 0" }}>
                <div
                  style={{
                    position: "absolute",
                    top: 4,
                    left: 4,
                    width: 16,
                    height: 16,
                    borderTop: `2px solid ${AMBER}`,
                    borderLeft: `2px solid ${AMBER}`,
                  }}
                />
                <div
                  style={{
                    position: "absolute",
                    top: 4,
                    right: 4,
                    width: 16,
                    height: 16,
                    borderTop: `2px solid ${AMBER}`,
                    borderRight: `2px solid ${AMBER}`,
                  }}
                />
                <div
                  style={{
                    position: "absolute",
                    bottom: 4,
                    left: 4,
                    width: 16,
                    height: 16,
                    borderBottom: `2px solid ${AMBER}`,
                    borderLeft: `2px solid ${AMBER}`,
                  }}
                />
                <div
                  style={{
                    position: "absolute",
                    bottom: 4,
                    right: 4,
                    width: 16,
                    height: 16,
                    borderBottom: `2px solid ${AMBER}`,
                    borderRight: `2px solid ${AMBER}`,
                  }}
                />
                <div style={{ display: "flex", alignItems: "center", gap: 8, padding: "0 22px" }}>
                  <Search size={18} color={INK_SOFT} />
                  <input
                    value={query}
                    onChange={(e) => {
                      setQuery(e.target.value);
                      if (searchState !== "idle") setSearchState("idle");
                      setMatched(null);
                    }}
                    onKeyDown={(e) => {
                      if (e.key === "Enter") runSearch(e.target.value);
                    }}
                    placeholder={`Digite ou leia o código (${idColumn})`}
                    style={{
                      flex: 1,
                      border: "none",
                      outline: "none",
                      background: "transparent",
                      fontFamily: "'IBM Plex Mono', monospace",
                      fontSize: 16,
                      padding: "10px 0",
                      color: INK,
                    }}
                  />
                  {query && (
                    <button
                      onClick={() => {
                        setQuery("");
                        setMatched(null);
                        setSearchState("idle");
                      }}
                      style={{
                        background: "transparent",
                        border: "none",
                        cursor: "pointer",
                        color: INK_SOFT,
                        display: "flex",
                      }}
                      aria-label="Limpar busca"
                    >
                      <X size={16} />
                    </button>
                  )}
                </div>
              </div>

              {query && suggestions.length > 0 && searchState === "idle" && (
                <div
                  style={{
                    border: `1px solid ${LINE}`,
                    borderRadius: 4,
                    marginTop: 6,
                    overflow: "hidden",
                  }}
                >
                  {suggestions.map((r, i) => (
                    <div
                      key={i}
                      onClick={() => {
                        setQuery(String(r[idColumn]));
                        runSearch(String(r[idColumn]));
                      }}
                      style={{
                        padding: "9px 12px",
                        fontSize: 13,
                        fontFamily: "'IBM Plex Mono', monospace",
                        cursor: "pointer",
                        borderBottom: i < suggestions.length - 1 ? `1px solid ${LINE}` : "none",
                        display: "flex",
                        justifyContent: "space-between",
                      }}
                      onMouseEnter={(e) => (e.currentTarget.style.background = PAPER)}
                      onMouseLeave={(e) => (e.currentTarget.style.background = "transparent")}
                    >
                      <span style={{ fontWeight: 600 }}>{String(r[idColumn])}</span>
                      {displayColumns[0] && (
                        <span style={{ color: INK_SOFT, fontFamily: "'IBM Plex Sans', sans-serif" }}>
                          {String(r[displayColumns[0]] ?? "")}
                        </span>
                      )}
                    </div>
                  ))}
                </div>
              )}

              {searchState === "notfound" && (
                <div
                  style={{
                    marginTop: 14,
                    display: "flex",
                    alignItems: "center",
                    gap: 8,
                    background: RED_BG,
                    color: RED,
                    border: `1px solid ${RED}`,
                    borderRadius: 4,
                    padding: "10px 12px",
                    fontSize: 13,
                  }}
                >
                  <AlertTriangle size={15} />
                  Nenhum item encontrado para "{query}" em {idColumn}.
                </div>
              )}

              {searchState === "found" && matched && (
                <div
                  style={{
                    marginTop: 16,
                    background: GREEN_BG,
                    border: `1px dashed ${GREEN}`,
                    borderRadius: 6,
                    padding: "16px 18px",
                  }}
                >
                  <div
                    style={{
                      display: "flex",
                      alignItems: "center",
                      gap: 8,
                      color: GREEN,
                      fontWeight: 600,
                      fontSize: 12.5,
                      textTransform: "uppercase",
                      letterSpacing: "0.04em",
                      marginBottom: 10,
                    }}
                  >
                    <Check size={15} /> Item localizado
                  </div>
                  <div
                    style={{
                      fontFamily: "'IBM Plex Mono', monospace",
                      fontSize: 20,
                      fontWeight: 600,
                      color: INK,
                      marginBottom: 10,
                    }}
                  >
                    {String(matched[idColumn])}
                  </div>
                  <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
                    {displayColumns.map((h) => (
                      <div
                        key={h}
                        style={{
                          display: "flex",
                          justifyContent: "space-between",
                          fontSize: 13.5,
                          borderTop: `1px solid ${LINE}`,
                          paddingTop: 6,
                        }}
                      >
                        <span style={{ color: INK_SOFT }}>{h}</span>
                        <span style={{ fontWeight: 500 }}>{String(matched[h] ?? "—")}</span>
                      </div>
                    ))}
                  </div>
                </div>
              )}
            </div>
          )}
        </Panel>

        {/* TABLE */}
        {hasData && (
          <Panel step={4} title={`Todos os itens (${rows.length})`} active={true}>
            <div style={{ overflowX: "auto" }}>
              <table style={{ width: "100%", borderCollapse: "collapse", fontSize: 12.5 }}>
                <thead>
                  <tr>
                    <th
                      style={{
                        textAlign: "left",
                        padding: "8px 10px",
                        borderBottom: `2px solid ${INK}`,
                        color: INK,
                        fontFamily: "'IBM Plex Mono', monospace",
                        whiteSpace: "nowrap",
                      }}
                    >
                      {idColumn}
                    </th>
                    {displayColumns.map((h) => (
                      <th
                        key={h}
                        style={{
                          textAlign: "left",
                          padding: "8px 10px",
                          borderBottom: `2px solid ${INK}`,
                          color: INK_SOFT,
                          fontWeight: 500,
                          whiteSpace: "nowrap",
                        }}
                      >
                        {h}
                      </th>
                    ))}
                  </tr>
                </thead>
                <tbody>
                  {rows.map((r, i) => {
                    const isMatch =
                      matched && String(r[idColumn]) === String(matched[idColumn]);
                    return (
                      <tr
                        key={i}
                        style={{
                          background: isMatch ? AMBER_BG : i % 2 === 0 ? PANEL : PAPER,
                        }}
                      >
                        <td
                          style={{
                            padding: "7px 10px",
                            borderBottom: `1px solid ${LINE}`,
                            fontFamily: "'IBM Plex Mono', monospace",
                            fontWeight: isMatch ? 600 : 400,
                            whiteSpace: "nowrap",
                          }}
                        >
                          {String(r[idColumn])}
                        </td>
                        {displayColumns.map((h) => (
                          <td
                            key={h}
                            style={{
                              padding: "7px 10px",
                              borderBottom: `1px solid ${LINE}`,
                              color: INK,
                              whiteSpace: "nowrap",
                            }}
                          >
                            {String(r[h] ?? "")}
                          </td>
                        ))}
                      </tr>
                    );
                  })}
                </tbody>
              </table>
            </div>
          </Panel>
        )}
      </div>
    </div>
  );
}
